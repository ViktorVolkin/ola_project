"use client";

import {useEffect, useRef} from "react";
import {usePathname, useRouter} from "next/navigation";
import {enableVisualEditing} from "@sanity/visual-editing";
import type {VisualEditingOptions} from "@sanity/visual-editing";
import {previewPath, previewNavigationPath, trustedStudio} from "@/content/sanity/preview-policy";

type Navigate = Parameters<
  NonNullable<VisualEditingOptions["history"]>["subscribe"]
>[0];

export function HomeVisualEditing() {
  const router = useRouter();
  const pathname = usePathname();
  const navigate = useRef<Navigate | undefined>(undefined);

  useEffect(() => {
    if (!previewPath(pathname)) return;

    const guard = (event: MessageEvent) => {
      if (!trustedStudio(event.origin)) {
        event.stopImmediatePropagation();
      }
    };

    window.addEventListener("message", guard, true);

    const media = window.matchMedia("(max-width: 599px)");

    const update = () => {
      document
        .querySelectorAll<HTMLElement>("[data-sanity-mobile]")
        .forEach((node) => {
          const value = media.matches
            ? node.dataset.sanityMobile
            : node.dataset.sanityDesktop;

          if (value) node.dataset.sanity = value;
        });
    };

    update();
    media.addEventListener("change", update);

    const observer = new MutationObserver(update);

    observer.observe(document.body, {
      childList: true,
      subtree: true,
    });

    const disable = enableVisualEditing({
      history: {
        subscribe(listener) {
          navigate.current = listener;
          listener({type: "push", url: location.href});

          return () => {
            navigate.current = undefined;
          };
        },

        update(event) {
          const path = previewNavigationPath(event.url, location.origin);
          if (!path) return;
          // Use the explicit validated destination, never an unbounded back().
          if (event.type === "replace" || event.type === "pop") {
            router.replace(path);
          } else {
            router.push(path);
          }
        },
      },

      refresh() {
        router.refresh();

        return new Promise<void>((resolve) =>
          setTimeout(resolve, 1000),
        );
      },
    });

    return () => {
      disable();
      window.removeEventListener("message", guard, true);
      media.removeEventListener("change", update);
      observer.disconnect();
    };
  }, [router, pathname]);

  useEffect(() => {
    navigate.current?.({
      type: "push",
      url: location.href,
    });
  }, [pathname]);

  return null;
}

export default async function LocaleLayout({
  children,
  params,
}: LayoutProps<"/[locale]">) {
  const {locale} = await params;

  if (!hasLocale(routing.locales, locale)) {
    notFound();
  }

  const [navigation, enquiry, draft] = await Promise.all([
    getTranslations("Navigation"),
    getTranslations("Enquiry"),
    draftMode(),
  ]);

  const visualEditingEnabled =
    draft.isEnabled &&
    Boolean(process.env.SANITY_API_READ_TOKEN) &&
    Boolean(process.env.SANITY_PREVIEW_ORIGIN);

  return (
    <html
      lang={languageTags[locale]}
      className={`${styles.root} ${manrope.variable} ${cormorant.variable} ${greatVibes.variable}`}
    >
      <body className={styles.body}>
        <a href="#main-content" className={styles.skipLink}>
          {navigation("skip")}
        </a>

        <NextIntlClientProvider locale={locale} messages={null}>
          <MotionScrollProvider>
            <Header />
            {children}

        </NextIntlClientProvider>

        <p
          id="enquiry-unavailable"
          className={styles.visuallyHidden}
        >
          {enquiry("unavailable")}
        </p>

        {visualEditingEnabled && <HomeVisualEditing />}
      </body>
    </html>
  );
}

export function previewPath(
  value: string | null
): value is string {
  if (!value) return false;

  const match = /^\/(en|ru|sr)(\/.*)?$/.exec(value);
  if (!match) return false;

  return previewRoutes.includes(
    (match[2] ?? "") as typeof previewRoutes[number]
  );
}

/** Reject normalization, encoding, credentials, query/hash and foreign-origin bypasses. */
export function previewNavigationPath(value: string, origin: string): string | null {
  try {
    const url = new URL(value, origin);
    return url.origin === origin && previewPath(url.pathname) &&
      (value === url.pathname || value === origin + url.pathname)
      ? url.pathname : null;
  } catch { return null; }
}
export function trustedStudio(origin: string | undefined): boolean {
  return !!origin && ((studioOrigins as readonly string[]).includes(origin) || (process.env.NODE_ENV === "development" && origin === "http://localhost:3333"));
}
/** Reject unknown/duplicate query parameters before the SDK can enable cookies. */
export function validEnableUrl(url: URL): boolean {
  const allowed = new Set(["sanity-preview-secret", "sanity-preview-pathname", "sanity-preview-perspective"]);
  for (const key of url.searchParams.keys()) if (!allowed.has(key) || url.searchParams.getAll(key).length !== 1) return false;
  return !!url.searchParams.get("sanity-preview-secret")?.trim() && homePath(url.searchParams.get("sanity-preview-pathname")) && [null, "drafts"].includes(url.searchParams.get("sanity-preview-perspective"));
}

import {isRedirectError} from "next/dist/client/components/redirect-error";
import {previewClient} from "@/content/sanity/preview-client";
import {trustedStudio, validEnableUrl} from "@/content/sanity/preview-policy";

export const dynamic = "force-dynamic";
const refuse = (status: number) => new Response("Preview unavailable", {status, headers: {"Cache-Control": "private, no-store", "Referrer-Policy": "no-referrer"}});
export async function GET(request: Request) {
  const client = previewClient();
  if (!client) return refuse(503);
  const url = new URL(request.url);
  if (url.origin !== process.env.SANITY_PREVIEW_ORIGIN || !validEnableUrl(url)) return refuse(400);
  try {
    // Extra origin restriction supplements, never replaces, the official secret check.
    const validated = await validatePreviewUrl(client, request.url);
    if (!validated.isValid || !trustedStudio(validated.studioOrigin)) return refuse(401);
    return await defineEnableDraftMode({client}).GET(request);
  } catch (error) {
    if (isRedirectError(error)) throw error;
    // Never log SDK errors: request objects can contain Authorization or preview secrets.
    return refuse(401);
  }
}
