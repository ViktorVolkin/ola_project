import {getLocale, getTranslations} from "next-intl/server";
import {getWeddingsContent} from "@/content/weddings";
import {resolveWeddingsContent} from "@/content/resolver/weddings";
import {weddingsEditing} from "@/components/widgets/WeddingsBlocks/WeddingsCommon/weddingsEditing";
import {WeddingsHeroBlock} from "@/components/widgets/WeddingsBlocks/WeddingsHeroBlock";
import {WeddingsCategoriesBlock} from "@/components/widgets/WeddingsBlocks/WeddingsCategoriesBlock";

export default async function WeddingsPage() {
  const [locale, t] = await Promise.all([getLocale(), getTranslations("Weddings")]);
  const {content, images, labels, source, preview} = await resolveWeddingsContent(locale);
  const local = getWeddingsContent(locale);
  const subject = local.hero.title.replace(/\n/g, " ");

  return (
    <main id="main-content" className={styles.page} data-locale={locale}>
      <WeddingsHeroBlock content={content.hero} previewStatus={source === "preview" ? {locale, preview, content} : undefined} image={images.hero} editing={editing} eventSubject={subject} homeLabel={labels.home ?? t("home")} planLabel={labels.planEvent ?? t("planEvent")} />

      <WeddingsApproachBlock content={content.approach} image={images.approach} editing={editing} processLabel={labels.ourProcess ?? t("ourProcess")} />
      <WeddingsProcessBlock content={content.process} editing={editing} />
      <WeddingsInspirationBlock
        content={content.inspiration}
        images={images}
        editing={editing}
        discussLabel={labels.discussEvent ?? t("discussEvent")}
        eventSubject={subject}
      />

        image={images.weddings}
        editing={editing}
        contactLabel={labels.contactWhatsapp ?? t("contactWhatsapp")}
        eventSubject={subject}

import "server-only";
import type {Locale} from "next-intl";
import {createPageEditing} from "@/content/sanity/editing";

export type WeddingsEditing = (
  path: string,
  image?: boolean
) => string | undefined;

export function weddingsEditing(
  locale: Locale,
  enabled: boolean
): WeddingsEditing {
  const editing = createPageEditing({
    documentId: "weddingsPage",
    documentType: "weddingsPage",
    locale,
    enabled,
  });

  return (path, image = false) =>
    image ? editing.image(path) : editing.text(path);
}

export function WeddingsEditable({
  attribute,
  children,
}: {
  attribute?: string;
  children: React.ReactNode;
}) {
  return attribute
    ? <span data-sanity={attribute}>{children}</span>
    : <>{children}</>;
}

export type ResolvedWeddings = CmsWeddings & {source: "default" | "sanity" | "preview"; preview: WeddingsPreview | null};
export async function resolveWeddingsContent(locale: Locale): Promise<ResolvedWeddings> {
  const source = (await draftMode()).isEnabled ? "preview" : contentSource(process.env.CONTENT_SOURCE);
  const local: CmsWeddings = {content: getWeddingsContent(locale), images: {}, labels: {}};
  if (source === "default") return {...local, source, preview: null};

    const value = await client.fetch<unknown>('*[_id == "weddingsPage" && _type == "weddingsPage"][0]', {}, {perspective: "drafts", cache: "no-store", stega: false, signal: AbortSignal.timeout(5000)});

    return {content: data.content, images: data.images, labels: data.labels, source, preview: {error: null, fallbackPaths: data.fallbackPaths}};

  SanityImageHotspot,

const locales = ["en", "ru", "sr"] as const;

type Obj = Record<string, unknown>;

const object = (value: unknown): Obj | null =>
  value !== null &&
  typeof value === "object" &&
  !Array.isArray(value)
    ? (value as Obj)
    : null;

const get = (value: unknown, path: readonly string[]): unknown =>
  path.reduce<unknown>(
    (current, key) => object(current)?.[key],
    value,
  );

type CanonicalImage = {
  asset: {_ref: string};
  crop?: SanityImageCrop;
  hotspot?: SanityImageHotspot;
};

/**
 * Accept only a validated Sanity image reference.
 * Do not forward arbitrary asset metadata to the image SDK.
 */
function canonicalImage(value: unknown): CanonicalImage | null {
  const input = object(value);
  if (!input) return null;

  const asset = object(input.asset);

  if (
    !asset ||
    "_ref" in input ||
    "_id" in input ||
    "url" in asset
  ) {
    return null;
  }

  const crop = object(input.crop);
  const hotspot = object(input.hotspot);

  const bounded = (entry: Obj, keys: string[]): boolean =>
    keys.every(
      (key) =>
        typeof entry[key] === "number" &&
        Number.isFinite(entry[key]) &&
        (entry[key] as number) >= 0 &&
        (entry[key] as number) <= 1,
    );

  if (
    input.crop !== undefined &&
    (
      !crop ||
      !bounded(crop, ["top", "bottom", "left", "right"]) ||
      (crop.left as number) + (crop.right as number) >= 1 ||
      (crop.top as number) + (crop.bottom as number) >= 1
    )
  ) {
    return null;
  }

  if (
    input.hotspot !== undefined &&
    (
      !hotspot ||
      !bounded(hotspot, ["x", "y", "width", "height"]) ||
      hotspot.width === 0 ||
      hotspot.height === 0
    )
  ) {
    return null;
  }

  const ref = get(value, ["asset", "_ref"]);

  if (typeof ref !== "string") return null;

  const match =
    /^image-([a-f0-9]{40})-([1-9]\d{0,4})x([1-9]\d{0,4})-(jpg|png|webp)$/.exec(ref);

  if (!match) return null;

  return {
    asset: {_ref: ref},
    ...(crop
      ? {
          crop: {
            top: crop.top as number,
            bottom: crop.bottom as number,
            left: crop.left as number,
            right: crop.right as number,
          },
        }
      : {}),
    ...(hotspot
      ? {
          hotspot: {
            x: hotspot.x as number,
            y: hotspot.y as number,
            width: hotspot.width as number,
            height: hotspot.height as number,
          },
        }
      : {}),
  };
}

 * Preserve the editor's crop and hotspot settings.
 */
function buildImageUrl(
  source: CanonicalImage,
  width: number,
): string | null {
  try {
    const url = imageUrlBuilder({
      projectId: "zhn2ddaj",
      dataset: "production",

      .image(source)
      .width(width)
      .url();

    const rect = new URL(url)

    }

    return url;
  } catch {
    return null;
  }
}

/**

 * the actual editor crop.
 */
export function imagePresentation(
  value: unknown,
): CmsImage | null {
  const source = canonicalImage(value);
  if (!source) return null;

  const src = buildImageUrl(source, 1600);
  if (!src) return null;

  const ref = source.asset._ref;
  const size = /-(\d+)x(\d+)-/.exec(ref);

  if (!size) return null;

  const width = Number(size[1]);
  const height = Number(size[2]);

  const rect =
    new URL(src)
      .searchParams
      .get("rect")
      ?.split(",")
      .map(Number) ??
    [0, 0, width, height];

  const hotspot = source.hotspot;

  const clamp = (value: number): number =>
    Math.max(0, Math.min(100, value));

  const x = clamp(
    (
      (Number(hotspot?.x ?? 0.5) * width - rect[0]) /
      rect[2]
    ) * 100,
  );

  const y = clamp(
    (
      (Number(hotspot?.y ?? 0.5) * height - rect[1]) /
      rect[3]
    ) * 100,
  );

  return {
    src,
    position: `${x}% ${y}%`,
  };
}

export function parseHome(
  value: unknown,
  locale: Locale,
): CmsHome | null {
  const doc = object(value);

  if (
    !doc ||
    doc._id !== "homePage" ||
    doc._type !== "homePage" ||
    !locales.includes(locale as (typeof locales)[number])
  ) {
    return null;
  }

  /**
   * Validate only the selected language.
   */
  const readText = (path: readonly string[]): string | null => {
    const text = get(doc, [...path, locale]);

    if (
      typeof text !== "string" ||
      !text.trim() ||
      text.length > 10000 ||
      /[<>]/.test(text)
    ) {
      return null;
    }

    return text;
  };

  /**
   * Start from complete local content.
   * Override only fields present and valid in Sanity.
   */
  const copy = structuredClone(getHomeContent(locale));

  const walk = (
    node: unknown,
    path: string[] = [],
  ): void => {
    const target = object(node);

    if (!target) return;

    for (const key of Object.keys(target)) {
      if (typeof target[key] === "string") {
        let sourcePath = [...path, key];

        /**
         * Local extras structure differs from Sanity:
         * extras.balloonAlt -> extras.balloons.alt.ru
         */
        if (
          path[0] === "extras" &&
          !["title", "description"].includes(key)
        ) {
          const map: Record<string, [string, string]> = {
            balloons: ["balloons", "title"],
            card: ["card", "title"],
            chocolates: ["chocolates", "title"],
            balloonAlt: ["balloons", "alt"],
            cardAlt: ["card", "alt"],
            chocolateAlt: ["chocolates", "alt"],
          };

          if (map[key]) {
            sourcePath = ["extras", ...map[key]];
          }
        }

        const text = readText(sourcePath);

        if (text !== null) {
          target[key] = text;
        }

        // Otherwise retain the local value.
      } else {
        walk(target[key], [...path, key]);
      }
    }
  };

  walk(copy);

  /**
   * Images can be added independently.
   * Unavailable images retain component-level fallback.
   */
  const images: CmsImages = {};

  for (
    const key of Object.keys(imagePaths) as
      (keyof typeof imagePaths)[]
  ) {
    const image = imagePresentation(
      get(doc, imagePaths[key]),
    );

    if (image) {
      images[key] = image;
    }
  }

  /**
   * Labels can also be added independently.
   * Missing labels use next-intl translations.
   */
  const labels: CmsLabels = {};

  for (const key of labelKeys) {
    const text = readText(["labels", key]);

    if (text !== null) {
      labels[key] = text;
    }
  }

  return {
    content: copy,
    images,
    labels,
  };
}

 * Fetch only published Sanity content.
 *
 * A missing or unavailable document returns null.
 * Individual missing fields no longer reject
 * the whole document.
 */
export async function fetchHome(locale: Locale) {
  const query = encodeURIComponent(
    '*[_id == "homePage" && _type == "homePage"][0]',
  );

  try {
    const response = await fetch(
      `https://zhn2ddaj.api.sanity.io/v2025-02-19/data/query/production?perspective=published&query=${query}`,
      {
        next: {revalidate: 60},
        signal: AbortSignal.timeout(3000),
      },
    );

    if (!response.ok) {
      throw new Error(`Sanity HTTP ${response.status}`);
    }

    const data: unknown = await response.json();

    const parsed = parseHome(
      object(data)?.result,
      locale,
    );

    if (!parsed) {
      console.warn(
        "Home CMS document unavailable; using local content",
        {locale},
      );
    }

    return parsed;
  } catch (error) {
    console.warn(
      "Home CMS request failed; using local content",
      {locale, error},
    );

    return null;
  }
}

function shape(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(shape);
  if (typeof value === "object" && value !== null) return Object.fromEntries(Object.entries(value).map(([key, entry]) => [key, shape(entry)]));
  expect(typeof value).toBe("string");
  expect(String(value).trim()).not.toBe("");
  return "string";
}
