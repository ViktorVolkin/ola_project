import type {Locale} from "next-intl";
import {getHomeContent} from "@/content/home";
import imageUrlBuilder from "@sanity/image-url";
import type {SanityImageSource} from "@sanity/image-url/lib/types/types";

const locales = ["en", "ru", "sr"] as const;
type Obj = Record<string, unknown>;
const object = (value: unknown): Obj | null => value !== null && typeof value === "object" && !Array.isArray(value) ? value as Obj : null;
const get = (value: unknown, path: string[]): unknown => path.reduce<unknown>((v, key) => object(v)?.[key], value);

/** Only Sanity asset references can produce URLs, never an editor-supplied URL. */
export function imageUrl(value: unknown, width = 1600): string | null {
  const input = object(value);
  if (!input) return null;
  const crop = object(input.crop);
  const hotspot = object(input.hotspot);
  const bounded = (entry: Obj, keys: string[]) => keys.every(key => typeof entry[key] === "number" && Number.isFinite(entry[key]) && (entry[key] as number) >= 0 && (entry[key] as number) <= 1);
  if (input.crop !== undefined && (!crop || !bounded(crop, ["top", "bottom", "left", "right"]) || (crop.left as number) + (crop.right as number) >= 1 || (crop.top as number) + (crop.bottom as number) >= 1)) return null;
  if (input.hotspot !== undefined && (!hotspot || !bounded(hotspot, ["x", "y", "width", "height"]) || hotspot.width === 0 || hotspot.height === 0)) return null;
  const ref = get(value, ["asset", "_ref"]);
  if (typeof ref !== "string") return null;
  const match = /^image-([a-f0-9]{40})-([1-9]\d{0,4})x([1-9]\d{0,4})-(jpg|png|webp)$/.exec(ref);
  if (!match) return null;
  try {
    // Width only: Sanity applies the editor's crop, never an estimated frame crop.
    // The browser alone performs the responsive cover using crop-relative focus.
    const url = imageUrlBuilder({projectId: "zhn2ddaj", dataset: "production"}).image(value as SanityImageSource).width(width).url();
    const rect = new URL(url).searchParams.get("rect")?.split(",").map(Number);
    return rect && (rect[2] < 1 || rect[3] < 1) ? null : url;
  } catch {
    return null;
  }
}

const imagePaths = {
  hero: ["hero", "photo"], flowers: ["services", "flowers", "photo"], balloons: ["services", "balloons", "photo"],
  weddings: ["services", "weddings", "photo"], weddingsMobile: ["services", "weddingsMobile"],
  romantic: ["moods", "romantic", "photo"], bright: ["moods", "bright", "photo"], natural: ["moods", "natural", "photo"],
  custom: ["custom", "photo"], extraBalloons: ["extras", "balloons", "photo"], card: ["extras", "card", "photo"],
  chocolates: ["extras", "chocolates", "photo"], meet: ["meet", "photo"], events: ["events", "photo"],
} as const;
export type CmsImage = {src: string; position: string};
export type CmsImages = Record<keyof typeof imagePaths, CmsImage>;

/** Normalize focus against the SDK's actual pixel-rounded editor crop.
 * Percentage object-position keeps that point at the same relative place in
 * every frame, including fluid and mobile frames; no viewport ratio guessing.
 * An oversized hotspot region cannot fit every cover frame; its center is kept.
 * Editor crop takes precedence if it excludes the hotspot center.
 */
export function imagePresentation(value: unknown): CmsImage | null {
  const src = imageUrl(value);
  if (!src) return null;
  const ref = get(value, ["asset", "_ref"]) as string;
  const [, width, height] = /-(\d+)x(\d+)-/.exec(ref)!;
  const rect = new URL(src).searchParams.get("rect")?.split(",").map(Number) ?? [0, 0, Number(width), Number(height)];
  const hotspot = object(object(value)?.hotspot);
  const clamp = (n: number) => Math.max(0, Math.min(100, n));
  const x = clamp(((Number(hotspot?.x ?? 0.5) * Number(width) - rect[0]) / rect[2]) * 100);
  const y = clamp(((Number(hotspot?.y ?? 0.5) * Number(height) - rect[1]) / rect[3]) * 100);
  return {src, position: `${x}% ${y}%`};
}
const labelKeys = ["shopBouquets", "letOlaChoose", "servicesLabel", "viewAllBouquets", "viewBouquet", "moodRailLabel", "letOlaCreate", "aboutOla", "exploreEvents"] as const;
export type CmsLabels = Record<typeof labelKeys[number], string>;

/** Reject incomplete documents as a whole: no cross-language or half-published mixing. */
export function parseHome(value: unknown, locale: Locale): {content: ReturnType<typeof getHomeContent>; images: CmsImages; labels: CmsLabels} | null {
  const doc = object(value);
  if (!doc || doc._id !== "homePage" || doc._type !== "homePage" || !locales.includes(locale as typeof locales[number])) return null;
  const readText = (path: string[]): string | null => {
    const field = get(doc, path);
    for (const lang of locales) {
      const text = get(field, [lang]);
      if (typeof text !== "string" || !text.trim() || text.length > 10000 || /[<>]/.test(text)) return null;
    }
    return get(field, [locale]) as string;
  };
  const copy = structuredClone(getHomeContent(locale));
  const walk = (node: unknown, path: string[] = []): boolean => {
    const target = object(node);
    if (!target) return false;
    for (const key of Object.keys(target)) {
      if (typeof target[key] === "string") {
        let sourcePath = [...path, key];
        if (path[0] === "extras" && !["title", "description"].includes(key)) {
          const map: Record<string, [string, string]> = {balloons: ["balloons", "title"], card: ["card", "title"], chocolates: ["chocolates", "title"], balloonAlt: ["balloons", "alt"], cardAlt: ["card", "alt"], chocolateAlt: ["chocolates", "alt"]};
          sourcePath = ["extras", ...map[key]];
        }
        const text = readText(sourcePath);
        if (text === null) return false;
        target[key] = text;
      } else if (!walk(target[key], [...path, key])) return false;
    }
    return true;
  };
  if (!walk(copy)) return null;
  const images = {} as CmsImages;
  for (const key of Object.keys(imagePaths) as (keyof CmsImages)[]) {
    const image = imagePresentation(get(doc, [...imagePaths[key]]));
    if (!image) return null;
    images[key] = image;
  }
  const labels = {} as CmsLabels;
  for (const key of labelKeys) {
    const text = readText(["labels", key]);
    if (text === null) return null;
    labels[key] = text;
  }
  return {content: copy, images, labels};
}

/** Public published perspective only; failure is visible in the server log. */
export async function fetchHome(locale: Locale) {
  const query = encodeURIComponent('*[_id == "homePage" && _type == "homePage"][0]');
  try {
    const response = await fetch(`https://zhn2ddaj.api.sanity.io/v2025-02-19/data/query/production?perspective=published&query=${query}`, {next: {revalidate: 60}, signal: AbortSignal.timeout(3000)});
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    const data: unknown = await response.json();
    const parsed = parseHome(object(data)?.result, locale);
    if (!parsed) console.warn("Home CMS unavailable or incomplete; using local content", {locale});
    return parsed;
  } catch (error) {
    console.warn("Home CMS request failed; using local content", {locale, error});
    return null;
  }
}
