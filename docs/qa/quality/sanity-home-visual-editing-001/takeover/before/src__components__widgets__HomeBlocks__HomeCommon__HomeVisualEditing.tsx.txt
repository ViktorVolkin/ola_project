"use client";
import {useCallback, useEffect} from "react";
import {usePathname, useRouter} from "next/navigation";
import {enableVisualEditing} from "@sanity/visual-editing";
import type {VisualEditingOptions} from "@sanity/visual-editing";
import {homePath, trustedStudio} from "@/content/sanity/preview-policy";

type Router = ReturnType<typeof useRouter>;
type Navigate = Parameters<NonNullable<VisualEditingOptions["history"]>["subscribe"]>[0];

let owners = 0;
let router: Router | null = null;
let navigate: Navigate | undefined;
let stop: (() => void) | undefined;

/** Draft-only island: official SDK overlays/comlink. Reload on draft mutation to refetch uncached draft data; no browser token. */
export function HomeVisualEditing() {
  const currentRouter = useRouter();
  const pathname = usePathname();
  const refresh = useCallback(() => {router?.refresh(); return Promise.resolve();}, [router]);
  useEffect(() => {
    owners++; router = currentRouter;
    const guard = (event: MessageEvent) => {
      // In Presentation, the parent window is the Studio. Trust messages from
      // parent/opener unconditionally; origin allowlisting protects non-parent
      // senders. frame-ancestors CSP + draft-mode enable route enforce Studio trust.
      const fromParent = event.source === window.parent || event.source === window.opener;
      if (!fromParent && !trustedStudio(event.origin)) event.stopImmediatePropagation();
    };
    window.addEventListener("message", guard, true);
    const media = window.matchMedia("(max-width: 599px)");
    const update = () => document.querySelectorAll<HTMLElement>("[data-sanity-mobile]").forEach(node => {
      const value = media.matches ? node.dataset.sanityMobile : node.dataset.sanityDesktop;
      if (value) node.dataset.sanity = value;
    });
    update(); media.addEventListener("change", update);
    const observer = new MutationObserver(update);
    observer.observe(document.body, {childList: true, subtree: true});
    const disable = enableVisualEditing({
      history: {
        subscribe(listener) {navigate = listener; listener({type: "push", url: location.href}); return () => {navigate = undefined;};},
        update(event) {
          if (event.type === "pop") {router?.back(); return;}
          const url = new URL(event.url, location.origin);
          if (url.origin !== location.origin || !homePath(url.pathname)) return;
          if (event.type === "replace") router?.replace(url.pathname); else router?.push(url.pathname);
        },
      },
      refresh,
    });
    return () => {
      owners--;
      disable();
      window.removeEventListener("message", guard, true);
      media.removeEventListener("change", update);
      observer.disconnect();
      if (!owners) {router = null; stop = undefined; navigate = undefined;}
    };
  }, [currentRouter, refresh]);
  useEffect(() => {navigate?.({type: "push", url: `${location.origin}${pathname}`});}, [pathname]);
  return null;
}
