'use client';

import { useState } from 'react';
import Image from 'next/image';
import styles from './BouquetDetailGallery.module.css';

export type GalleryImage = Readonly<{
  src: string;
  alt: string;
  width: number;
  height: number;
  thumbnailLabel: string;
}>;

type Props = Readonly<{
  images: readonly [GalleryImage, ...GalleryImage[]];
  groupLabel: string;
}>;

export function BouquetDetailGallery({ images, groupLabel }: Props) {
  const [selected, setSelected] = useState(0);
  const current = images[selected];

  return (
    <section className={styles.gallery} aria-label={groupLabel}>
      <Image className={styles.main} src={current.src} alt={current.alt} width={current.width} height={current.height} sizes="(min-width: 1440px) 624px, (min-width: 1100px) calc(45.02vw - 24px), (min-width: 768px) 704px, calc(100vw - 40px)" loading="eager" />
      <div className={styles.thumbnails} role="group" aria-label={groupLabel}>
        {images.map((image, index) => (
          <button key={image.src} className={styles.thumbnail} type="button" aria-label={image.thumbnailLabel} aria-pressed={selected === index} onClick={() => setSelected(index)}>
            <Image src={image.src} alt="" width={image.width} height={image.height} sizes="(min-width: 1440px) 144px, (min-width: 1100px) calc(11.25vw - 18px), (min-width: 768px) 164px, calc(25vw - 22px)" />
          </button>
        ))}
      </div>
    </section>
  );
}
