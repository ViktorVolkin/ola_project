function classify(file,tag,node) {
 if (['ActionLink','ActionButton','EnquiryLink'].includes(tag) || file.includes('/shared/ActionLink.tsx') || file.includes('/shared/ActionButton.tsx') || file.includes('/EnquiryLink/')) return 'chrome-action';
 if(file.includes('/Header/DesktopNavigation')) return 'chrome-desktop-nav';
 if(file.includes('/Header/MobileNavigation')) return tag==='Link'?'chrome-mobile-nav':'chrome-mobile-controls';
 if(file.includes('/Footer/')) return 'chrome-footer-nav';
 if(file.includes('/LanguageSwitcher/')) return 'chrome-language';
 if(file.includes('/Wordmark/')) return 'chrome-wordmark';
 if(file.includes('/app/') && tag==='a') return 'page-skip';
 if (node.getText().includes('actionClassName(')) return 'chrome-action';
 if(file.includes('/BouquetDetailHero/')) return node.getText().includes('styles.back')?'page-detail-back':'page-breadcrumb';
 if(['/BouquetsHero/','/BalloonsHeroBlock.tsx','/WeddingsHeroBlock/','/ContactHero.tsx'].some(x=>file.includes(x))) return 'page-breadcrumb';
 if(file.includes('/HomeMoodBlock/') && tag==='div') return 'page-scroll-rail';
 if(file.includes('/HomeMoodBlock/') || file.includes('/HomeServicesBlock/')) return 'page-home-cards';
 if(file.includes('/BouquetsInspiration/')) return tag==='button'?'page-bouquet-disabled':'page-bouquet-selectors';
 if(file.includes('/BouquetDetailGallery/')) return 'page-gallery';
 if(file.includes('/BouquetDetailPreferences/')) return 'page-preferences';
 if(file.includes('/BouquetDetailSupporting/') || file.includes('/ContactFaq.tsx')) return 'page-disclosures';
 return 'review-unclassified';
}

 if(['a','button','motion.button','Dialog.Close','Link','ActionLink','ActionButton','EnquiryLink','input','select','textarea','summary'].includes(tag)||hasClick||interactiveRole||tabStop){const rel=path.relative(root,file);const pos=sf.getLineAndCharacterOfPosition(n.getStart(sf));const line=pos.line+1;const column=pos.character+1;const id=classify('/'+rel,tag,n);families[id].occurrences.push({path:rel,line,column,tag,opening:n.getText(sf).replace(/\s+/g,' ').slice(0,500)});}}
