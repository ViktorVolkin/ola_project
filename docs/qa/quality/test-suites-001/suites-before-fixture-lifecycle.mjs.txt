#!/usr/bin/env node
// Builds private source exports. Never reuses an unverified build or a listening server.
import {cpSync, mkdirSync, mkdtempSync, readFileSync, readdirSync, writeFileSync, symlinkSync, realpathSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
import path from 'node:path';
import {listInventory} from './suite-inventory.mjs';
import {selection, playwrightArgs, runGroups} from './suite-controls.mjs';

const root = process.cwd();
const {mode, options, selected} = selection(process.argv.slice(2));
if (mode === 'inventory') { const result=listInventory(root); console.log(JSON.stringify(result,null,2)); process.exit(result.unknown.length || result.missing.length ? 1 : 0); }
const report=listInventory(root);
if (report.unknown.length || report.missing.length) {
  const results=(mode === 'full' ? ['unit',...selected] : selected).map(suite=>({suite,status:'BLOCKED',error:'inventory guard failed'}));
  console.log('FULL SUMMARY',JSON.stringify({results,inventory:report},null,2));
  process.exit(1);
}
const port=Number(process.env.OLA_SUITE_PORT || '45190');
if (!Number.isInteger(port) || port < 1024 || port > 65535 || [3100,42242,42244].includes(port)) throw new Error('Invalid or reserved port');
if (options['--list'] || options['--dry-run']) {
  console.log(JSON.stringify({mode,selected,options,port,inventory:report},null,2)); process.exit(0);
}
const allowedRoot=path.join(root,'.local-tools/ola-qa/test-suite-runs');
const parent=path.resolve(process.env.OLA_SUITE_RUN_ROOT || allowedRoot);
if (parent !== allowedRoot && !parent.startsWith(allowedRoot+path.sep)) throw new Error('OLA_SUITE_RUN_ROOT must be within .local-tools/ola-qa/test-suite-runs');
mkdirSync(parent,{recursive:true});
if (realpathSync(parent) !== path.join(realpathSync(root),path.relative(root,parent))) throw new Error('Run root must not traverse symlinks');
const run=mkdtempSync(path.join(parent,'run-'));
const work=path.join(run,'source'); mkdirSync(work);
console.log(`Task-owned run: ${run}`);
const excluded = new Set(['node_modules','.next','test-results','.git','.local-tools','.npmrc']);
function allowed(src) {const b=path.basename(src); return !excluded.has(b) && !b.startsWith('.env');}
function copy(name) {mkdirSync(path.dirname(path.join(work,name)),{recursive:true}); cpSync(path.join(root,name),path.join(work,name),{recursive:true,filter:allowed,errorOnExist:true,force:false});}
function manifest(dir, result={}) {
  for (const entry of readdirSync(dir,{withFileTypes:true})) {
    const f=path.join(dir,entry.name);
    if (entry.isDirectory()) manifest(f,result);
    else if (entry.isFile()) result[path.relative(work,f)] = createHash('sha256').update(readFileSync(f)).digest('hex');
    else throw new Error(`Unexpected symbolic link in input export: ${f}`);
  }
  return result;
}
let before;
try {
  for (const p of ['src','public','tests','package.json','package-lock.json','next.config.ts','tsconfig.json','next-env.d.ts','vitest.config.mts','scripts/model-quality-summary.mjs','docs/assets-manifest.json','docs/design/balloons-intake.json']) copy(p);
  before=manifest(work); writeFileSync(path.join(run,'inputs.json'),JSON.stringify(before,null,2));
  for (const [file,hash] of Object.entries(before)) if (createHash('sha256').update(readFileSync(path.join(root,file))).digest('hex') !== hash) throw new Error(`Input changed during export: ${file}`);
  symlinkSync(path.join(root,'node_modules'),path.join(work,'node_modules'),'dir');
} catch (error) {
  const results=(mode === 'full' ? ['unit',...selected] : selected).map(suite=>({suite,status:'BLOCKED',error:String(error)}));
  writeFileSync(path.join(run,'summary.json'),JSON.stringify({run,results},null,2)+'\n');
  console.log('FULL SUMMARY',JSON.stringify(results,null,2)); process.exit(1);
}
const next=path.join(work,'node_modules/next/dist/bin/next');
const pw=path.join(work,'node_modules/@playwright/test/cli.js');
const env=Object.fromEntries(['PATH','HOME','XDG_CONFIG_HOME','XDG_CACHE_HOME','SHELL','SWC_NATIVE_BINDING_CACHE','NODE_DISABLE_COMPILE_CACHE','PLAYWRIGHT_BROWSERS_PATH','CI'].filter(key=>process.env[key] !== undefined).map(key=>[key,process.env[key]]));
Object.assign(env,{OLA_SUITE_PORT:String(port), OLA_SUITE_BASE_URL:`http://127.0.0.1:${port}`, BD_MIGRATION_EVIDENCE:path.join(run,'migration-evidence')});
mkdirSync(env.BD_MIGRATION_EVIDENCE,{recursive:true});
function execute(command, argv, cwd, extra={}) {
  console.log(`\n$ ${JSON.stringify([command,...argv])} [cwd ${cwd}]`);
  const out=spawnSync(command,argv,{cwd,env:{...env,...extra},stdio:'inherit'});
  return out.error ? {code:null,error:String(out.error)} : {code:out.status, signal:out.signal};
}
function appCopy(name) {
  const app=path.join(run,name); mkdirSync(app);
  for (const p of ['src','public','package.json','package-lock.json','next.config.ts','tsconfig.json','next-env.d.ts']) cpSync(path.join(work,p),path.join(app,p),{recursive:true});
  symlinkSync(path.join(root,'node_modules'),path.join(app,'node_modules'),'dir');
  return app;
}
const aggregate=runGroups(mode, selected, () => {
  const unit=execute(process.execPath,[path.join(work,'node_modules/vitest/vitest.mjs'),'run'],work);
  return {suite:'unit',status:unit.code===0?'PASS':'FAIL',...unit};
}, family => {
  let result;
  try {
    let app=work;
    if (family === 'detail-page') {
      app=appCopy('detail-page-app');
      const route=path.join(app,'src/app/[locale]/fixture/detail-page'); mkdirSync(route,{recursive:true});
      cpSync(path.join(work,'tests/fixtures/bouquet-detail-page/page.tsx.txt'),path.join(route,'page.tsx'));
    } else if (family === 'migration') {
      app=appCopy('migration-app');
      const route=path.join(app,'src/app/[locale]/fixture/detail'); mkdirSync(route,{recursive:true});
      cpSync(path.join(work,'tests/fixtures/bouquet-detail-next/migration-page.tsx.txt'),path.join(route,'page.tsx'));
      const block=(file,start,end)=>{const text=readFileSync(path.join(work,file),'utf8'); const a=text.indexOf(start), b=text.indexOf(end,a); if (a<0 || b<0) throw new Error(`Fixture data markers absent in ${file}`); return text.slice(a,b);};
      const prefs=block('tests/e2e/bouquet-detail-preferences.spec.ts','const labels = {','\nconst extraImages');
      const support=block('tests/e2e/bouquet-detail-supporting.spec.ts','const content = {','\nconst files=');
      const gallery=block('tests/e2e/bouquet-detail-gallery.spec.ts','const labels = {','\nlet script').replace('const labels =','const galleryLabels =');
      writeFileSync(path.join(route,'fixture-data.ts'),`${prefs}\n${support}\n${gallery}\nexport {labels, content, galleryLabels};\n`);
    } else if (family === 'action-button') app=path.join(work,'tests/fixtures/action-button');
    if (family !== 'gallery') {
      const build=execute(process.execPath,[next,'build','--webpack'],app);
      if (build.code !== 0) throw new Error(`fixture build failed ${JSON.stringify(build)}`);
      const id=readFileSync(path.join(app,'.next/BUILD_ID'),'utf8').trim();
      console.log(`${family} freshly built BUILD_ID=${id}`);
    }
    const jsonReport=path.join(run,`${family}-playwright.json`);
    const cli=playwrightArgs(work,options);
    const outcome=execute(process.execPath,[pw,...cli],work,{OLA_SUITE_FAMILY:family,OLA_SUITE_APP:app,PLAYWRIGHT_JSON_OUTPUT_NAME:jsonReport});
    let stats;
    try { stats=JSON.parse(readFileSync(jsonReport,'utf8')).stats; } catch { stats=null; }
    result={suite:family,status:outcome.code===0 && stats ? 'PASS':'FAIL',...outcome,stats,report:jsonReport};
    try {
      const changed=[];
      for (const [file,hash] of Object.entries(before)) if (createHash('sha256').update(readFileSync(path.join(root,file))).digest('hex') !== hash) changed.push(file);
      if (changed.length) throw new Error(`Root input changed during ${family}: ${changed.join(',')}`);
    } catch (error) { result={...result,status:'FAIL',error:String(error)}; }
  } catch (error) { result={suite:family,status:'BLOCKED',error:String(error)}; }
  return result;
}, result => console.log('SUITE',JSON.stringify(result)));
const {results,exitCode}=aggregate;
writeFileSync(path.join(run,'summary.json'),JSON.stringify({run,results},null,2)+'\n');
console.log('\nFULL SUMMARY',JSON.stringify(results,null,2));
process.exitCode=exitCode;
