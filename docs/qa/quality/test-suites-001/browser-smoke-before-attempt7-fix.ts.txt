import {expect, test} from '@playwright/test';
import {mkdir} from 'node:fs/promises';
test('real disabled EnquiryLink across locales and viewport geometry', async ({page}) => {
  for (const locale of ['en', 'ru', 'sr']) {
    await page.setViewportSize({width: 360, height: 800});
    await page.goto(`/${locale}`);
    const buttons = page.locator('[data-compare="legacy"] button, [data-compare="adopted"] button');
    await expect(buttons).toHaveCount(2);
    const legacy = buttons.nth(0), adopted = buttons.nth(1);
    for (const button of [legacy, adopted]) {
      await expect(button).toHaveAttribute('type', 'button');
      await expect(button).toBeDisabled();
      await expect(button).toHaveAccessibleName(locale === 'ru' ? 'Спросить студию' : locale === 'sr' ? 'Pitajte studio' : 'Ask the studio');
      await expect(button.locator('svg')).toBeVisible();
      await expect(button.locator('span')).toBeVisible();
    }
    await expect(adopted).not.toHaveAttribute('href', /.+/);
    await expect(adopted).toHaveAttribute('aria-describedby', 'enquiry-unavailable');
    await expect(page.locator('#enquiry-unavailable')).toContainText(locale === 'ru' ? 'Обращения в WhatsApp пока недоступны' : locale === 'sr' ? 'WhatsApp upiti još nisu dostupni' : 'WhatsApp enquiries are not available yet');
    const measure = async () => Promise.all([legacy, adopted].map(async button => button.evaluate(el => {const r=el.getBoundingClientRect(),s=getComputedStyle(el);return {x:r.x,y:r.y,width:r.width,height:r.height,display:s.display,font:s.font,padding:s.padding,border:s.border,bg:s.backgroundColor,color:s.color,gap:s.gap,align:s.alignItems,justify:s.justifyContent}})));
    const small = await measure();
    expect(small[1]).toEqual(small[0]);
    await page.setViewportSize({width: 1440, height: 900});
    const wide = await measure();
    expect(wide[1]).toEqual(wide[0]);
    const evidence = {locale, small, wide};
    await mkdir('evidence', {recursive: true});
    await page.screenshot({path: `evidence/${locale}-1440.png`, fullPage: true});
    console.log(JSON.stringify(evidence));
  }
});
