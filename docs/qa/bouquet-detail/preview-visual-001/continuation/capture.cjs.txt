const { chromium } = require('@playwright/test');
const fs = require('fs');
const path = require('path');
(async () => {
  const dir = __dirname;
  const browser = await chromium.launch({headless: true});
  const result = {};
  for (const width of [360, 768, 1440]) {
    const page = await browser.newPage({viewport:{width, height:900}, deviceScaleFactor:1});
    await page.goto('http://127.0.0.1:3223/en/preview/bouquet', {waitUntil:'networkidle'});
    await page.evaluate(() => document.fonts.ready);
    const data = await page.evaluate(() => {
      const img = document.querySelector('img[src="/assets/decorative/story_botanical_transparent.png"]');
      const band = img.parentElement;
      const h1 = document.querySelector('main h1');
      const eyebrow = h1.previousElementSibling;
      const box = el => { if (!el) return null; const r=el.getBoundingClientRect(); return {x:r.x+scrollX,y:r.y+scrollY,width:r.width,height:r.height}; };
      const css = el => {const c=getComputedStyle(el); return {fontFamily:c.fontFamily,fontSize:c.fontSize,fontWeight:c.fontWeight,lineHeight:c.lineHeight,letterSpacing:c.letterSpacing,display:c.display,background:c.backgroundColor,objectFit:c.objectFit};};
      return {band:box(band),image:box(img),introHeading:box(h1),headingStyle:css(h1),eyebrowStyle:css(eyebrow),bandStyle:css(band),imageStyle:css(img),footerCount:document.querySelectorAll('footer').length,documentHeight:document.documentElement.scrollHeight};
    });
    result[width]=data;
    await page.screenshot({path:path.join(dir, `viewport-${width}.png`)});
    await page.locator('img[src="/assets/decorative/story_botanical_transparent.png"]').first().scrollIntoViewIfNeeded();
    await page.screenshot({path:path.join(dir, `botanical-viewport-${width}.png`)});
    await page.locator('img[src="/assets/decorative/story_botanical_transparent.png"]').first().locator('..').screenshot({path:path.join(dir, `botanical-crop-${width}.png`)});
    await page.close();
  }
  fs.writeFileSync(path.join(dir,'browser-rects.json'), JSON.stringify(result,null,2)+'\n');
  await browser.close();
})().catch(error => {console.error(error);process.exit(1);});
