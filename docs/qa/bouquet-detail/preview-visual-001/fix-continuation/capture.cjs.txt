const { chromium } = require('/workspace/node_modules/@playwright/test');
const fs = require('fs');
(async () => {
 const browser=await chromium.launch({headless:true}); const all={};
 for(const width of [360,768,1440]) {
  const page=await browser.newPage({viewport:{width,height:900},deviceScaleFactor:1});
  await page.goto('http://127.0.0.1:3223/en/preview/bouquet',{waitUntil:'networkidle'}); await page.evaluate(()=>document.fonts.ready);
  all[width]=await page.evaluate(()=>[...document.querySelectorAll('main section:first-of-type, main section:first-of-type *')].filter(e=>{const r=e.getBoundingClientRect();return r.width>0 && r.height>0 && (e.children.length || ['P','H1','H2','H3','SUMMARY','LABEL'].includes(e.tagName))}).map(e=>{const r=e.getBoundingClientRect(),c=getComputedStyle(e); return {tag:e.tagName,cls:e.className,text:(e.innerText||'').slice(0,160),x:r.x+scrollX,y:r.y+scrollY,w:r.width,h:r.height,font:c.fontFamily,size:c.fontSize,line:c.lineHeight,weight:c.fontWeight,gap:c.gap,margin:c.margin,padding:c.padding}}));
  await page.screenshot({path:`${__dirname}/pristine-${width}.png`,fullPage:true}); await page.close();
 }
 fs.writeFileSync(`${__dirname}/browser-blocks.json`,JSON.stringify(all,null,2)); await browser.close();
})().catch(e=>{console.error(e);process.exit(1)});
