const {chromium, expect} = require('/workspace/node_modules/@playwright/test');
(async () => {
  const browser = await chromium.launch({headless: true});
  try {
    for (const locale of ['en','ru','sr']) for (const mobile of [false,true]) {
      const page = await browser.newPage({viewport: mobile ? {width:375,height:812} : {width:1440,height:900}});
      await page.goto(`http://127.0.0.1:3100/${locale}/about?source=qa#main-content`);
      if (mobile) await page.locator('header button[aria-haspopup="dialog"]').click();
      const root = mobile ? page.getByRole('dialog') : page.getByRole('banner');
      const trigger = mobile ? root.locator('button[aria-controls]').last() : root.locator('button[aria-controls]').first();
      await trigger.click();
      await expect(trigger).toHaveAttribute('aria-expanded','true');
      const links = root.locator('ul[id] a');
      await expect(links).toHaveCount(3);
      await links.nth(1).focus();
      await page.keyboard.press('Shift+Tab');
      await expect(links.first()).toBeFocused();
      await expect(trigger).toHaveAttribute('aria-expanded','true');
      await trigger.focus();
      await page.keyboard.press('Shift+Tab');
      await expect(trigger).toHaveAttribute('aria-expanded','false');
      await expect(trigger).not.toBeFocused();
      if (mobile) { await expect(root).toBeVisible(); await expect(page.locator('body')).toHaveCSS('overflow','hidden'); }
      await trigger.click();
      const target = {en:'Srpski',ru:'English',sr:'Русский'}[locale];
      const expected = {en:'sr',ru:'en',sr:'ru'}[locale];
      await root.getByRole('link',{name:target,exact:true}).click();
      await expect(page).toHaveURL(new RegExp(`/${expected}/about\\?source=qa#main-content$`));
      console.log(`PASS ${locale} ${mobile?'mobile':'desktop'} reverse boundary, internal focus, locale route/query/hash`);
      await page.close();
    }
  } finally { await browser.close(); }
})().catch(error => {console.error(error); process.exitCode=1;});
